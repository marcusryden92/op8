// OF2 suit boot sequence. Plays when the suit comes online, or with the
// suit_bootsequence console command, which reloads this file every time.
//
// Settings:
//	start_delay		seconds before the first line
//	max_lines		lines on screen before older ones scroll up and fade out
//	bar_column		character column the bars start at (keeps them lined up)
//	bar_width		bar length in "=" characters
//	type_rate		default typing speed, characters per second
//	scroll_speed	minimum lines per second when scrolling (it speeds up to keep up)
//	age_fade		how much the oldest visible line is dimmed (0..1)
//	end_hold		seconds the finished text stays up
//	end_fade		seconds to fade it out
//	burst_rate		default lines per second for bursts
//	burst_alpha		default brightness of burst lines (0..1)
//	burst_words		space-separated words for {word} / {WORD}
//	burst_paths		space-separated paths for {path}
//	burst_sets		named groups of templates; each "t" value is one template.
//					A set may also have its own "words" and "paths", which
//					replace burst_words / burst_paths for that set only.
//					Placeholders: {word} {WORD} {path} {hex} {hex4} {kb}
//					{num} (0-9999) and {num:MIN-MAX} (inclusive; a malformed
//					range is printed literally so typos show up on screen)
//
// Each "line" is either a text line:
//	"text"			the text, or a "#Localization_Token"
//	"type_rate"		typing speed for this line (0 = appears at once)
//	"bar"			seconds to fill a progress bar (leave out for no bar)
//	"stall"			"<bar fraction> <seconds>": the bar freezes there. Repeat up to 4
//					times per line, in increasing order.
//	"result"		printed after "; " when the line finishes
//	"result_color"	"normal", "caution" (amber) or "critical" (red)
//	"color"			text and bar color: "normal", "caution" or "critical"
//	"alpha"			brightness (0..1, default 1.0)
//	"command"		console command run when the line finishes (when the result
//					prints, for bar lines). Several can be separated by ";".
//	"pause"			seconds before the next line starts
// or a burst of random lines from a template set:
//	"burst"			how many lines
//	"set"			which burst_sets group (default "default")
//	"rate"			lines per second for this burst
//	"color"			color of the burst lines
//	"alpha"			brightness of the burst lines (default burst_alpha)
//	"command"		console command run after the last line, before "pause"
//	"pause"			stop after the burst
//
// Blank "text" makes an empty line. Lines still dim with age on top of "alpha".
// Tip: a text line with "type_rate" "0" and "alpha" "0.55" looks like a burst
// line but only appears once; used below for one-off asides.
